Понял задачу. Нужен бот-админка для розыгрышей в канале: создание лота, условия, дедлайн, выбор победителей + капча на входе с примером и инлайн-кнопками.

Сразу скажу: в результатах поиска есть только общие куски по капче на telebot и настройке инлайн-кнопок, но готового кода под полноценный бот розыгрышей нет. Поэтому ниже — рабочая структура и ключевые куски кода, которые можно собрать в проект.

Стек и логика
Библиотека: pyTelegramBotAPI (telebot) — проще всего для старта, есть примеры инлайн-капчи .

Хранение данных: SQLite (розыгрыши, участники, настройки). Для старта можно JSON, но при розыгрышах в канале лучше сразу БД.

Роли: админ (создаёт розыгрыш) / участник (жмёт кнопку «Участвовать»).

Капча: показывается один раз при /start, после прохождения пользователь допускается к участию.

1. Капча с примером и инлайн-кнопками
Идея: генерируем пример a + b, правильный ответ и 4 неверных. В callback_data кладём captcha:<правильный>:<выбранный> — обработчик сравнивает.

python
import random
from telebot import types

def generate_captcha():
    a = random.randint(1, 10)
    b = random.randint(1, 10)
    correct = a + b
    wrong = set()
    while len(wrong) < 4:
        wrong.add(random.randint(2, 20))
    wrong.discard(correct)
    options = list(wrong) + [correct]
    random.shuffle(options)
    return a, b, correct, options

@bot.message_handler(commands=['start'])
def start_cmd(message):
    # проверка, проходил ли уже капчу (по user_id в БД)
    if is_verified(message.from_user.id):
        show_main_menu(message.chat.id)
        return

    a, b, correct, options = generate_captcha()
    markup = types.InlineKeyboardMarkup(row_width=2)
    buttons = []
    for opt in options:
        buttons.append(types.InlineKeyboardButton(
            str(opt),
            callback_data=f"cap:{correct}:{opt}"
        ))
    markup.add(*buttons)

    bot.send_message(
        message.chat.id,
        f"🧮 Реши пример, чтобы продолжить:\n\n<b>{a} + {b} = ?</b>",
        parse_mode='HTML',
        reply_markup=markup
    )

@bot.callback_query_handler(func=lambda call: call.data.startswith('cap:'))
def handle_captcha(call):
    _, correct, chosen = call.data.split(':')
    if int(chosen) == int(correct):
        mark_verified(call.from_user.id)
        bot.answer_callback_query(call.id, "✅ Верно!")
        bot.edit_message_text(
            "Капча пройдена. Добро пожаловать!",
            chat_id=call.message.chat.id,
            message_id=call.message.message_id
        )
        show_main_menu(call.message.chat.id)
    else:
        bot.answer_callback_query(call.id, "❌ Неверно, попробуй ещё раз", show_alert=True)
Ключевой момент: callback_data не должен превышать 64 байта — cap:12:8 влезает спокойно.

6. Что учесть, чтобы не сломалось
bot.answer_callback_query обязателен на каждое нажатие кнопки, иначе у пользователя крутится спиннер .

Капча должна быть один раз, а не при каждом /start. Храни verified в БД по user_id.

Проверка подписки на канал (если условие «быть подписанным»): используй bot.get_chat_member(channel_id, user_id) перед добавлением в участники.

Не храни состояние в памяти — при перезапуске всё потеряется. SQLite или JSON-файл.

Callback data ≤ 64 байта — не пихай туда длинные тексты.

Сразу оговорюсь: в результатах поиска есть только разрозненные примеры по answer_callback_query, get_chat_member и инлайн-кнопкам, но готового «красивого» шаблона под такой бот нет. Поэтому собираем сами, но по-взрослому.

1. Капча: один раз и навсегда
В таблице users держим флаг verified:

python
# database.py
import sqlite3

conn = sqlite3.connect('bot.db', check_same_thread=False)
cur = conn.cursor()

cur.execute("""
CREATE TABLE IF NOT EXISTS users (
    user_id INTEGER PRIMARY KEY,
    verified INTEGER DEFAULT 0,
    username TEXT
)
""")
conn.commit()

def is_verified(user_id: int) -> bool:
    cur.execute("SELECT verified FROM users WHERE user_id = ?", (user_id,))
    row = cur.fetchone()
    return bool(row and row[0])

def mark_verified(user_id: int, username: str = ""):
    cur.execute("""
        INSERT INTO users (user_id, verified, username) VALUES (?, 1, ?)
        ON CONFLICT(user_id) DO UPDATE SET verified = 1
    """, (user_id, username))
    conn.commit()
Капча генерируется только если is_verified вернул False. После успешного ответа флаг ставится в 1, и больше пользователь её не увидит — даже если нажмёт /start повторно .

@bot.message_handler(func=lambda m: m.from_user.id in drafts and drafts[m.from_user.id]['step'] == 'end_at')
def step_end_at(message):
    try:
        dt = datetime.datetime.strptime(message.text.strip(), "%Y-%m-%d %H:%M")
    except ValueError:
        bot.send_message(message.chat.id, "Формат неверный. Пример: 2026-06-01 20:00")

    # публикуем пост в канал
    markup = types.InlineKeyboardMarkup()

        f"🎁 <b>Розыгрыш: {d['prize']}</b>\n\n"
        f"{d['conditions']}\n\n"
        f"👥 Победителей: {d['winners']}\n"
        f"⏰ Итоги: {dt.strftime('%d.%m.%Y %H:%M')}"
    )
    msg = bot.send_message(CHANNEL_ID, text, parse_mode='HTML', reply_markup=markup)

    # сохраняем в БД и обновляем callback_data на реальный id
    cur.execute(
        "INSERT INTO giveaways (channel_id, message_id, prize, winners_count, conditions, end_at) VALUES (?,?,?,?,?,?)",
        (CHANNEL_ID, msg.message_id, d['prize'], d['winners'], d['conditions'], dt.isoformat())
    )
    conn.commit()
    gid = cur.lastrowid

    # правим кнопку на join:gid
    new_markup = types.InlineKeyboardMarkup()
    new_markup.add(types.InlineKeyboardButton("🎉 Участвовать", callback_data=f"join:{gid}"))
    bot.edit_message_reply_markup(CHANNEL_ID, msg.message_id, reply_markup=new_markup)

    bot.send_message(message.chat.id, f"✅ Розыгрыш #{gid} опубликован.")
Важный момент: кнопка «Участвовать» в канале должна иметь реальный id розыгрыша. Поэтому сначала отправляем с заглушкой, потом правим reply_markup через edit_message_reply_markup.

# giveaway.py
import random

@bot.callback_query_handler(func=lambda call: call.data.startswith('join:'))
def join(call):
    gid = int(call.data.split(':')[1])
    uid = call.from_user.id

    if not is_verified(uid):
        bot.answer_callback_query(call.id, "Сначала пройди капчу в боте: /start", show_alert=True)
        return

    cur.execute("SELECT status FROM giveaways WHERE id = ?", (gid,))
    row = cur.fetchone()
    if not row or row[0] != 'active':
        bot.answer_callback_query(call.id, "Розыгрыш уже завершён", show_alert=True)
        return

    cur.execute("SELECT 1 FROM participants WHERE user_id = ? AND giveaway_id = ?", (uid, gid))
    if cur.fetchone():
        bot.answer_callback_query(call.id, "Ты уже участвуешь", show_alert=True)
        return

    cur.execute("INSERT INTO participants (user_id, giveaway_id) VALUES (?,?)", (uid, gid))
    conn.commit()
    bot.answer_callback_query(call.id, "✅ Ты в списке участников!")


def pick_winners(gid):
    cur.execute("SELECT winners_count FROM giveaways WHERE id = ?", (gid,))
    n = cur.fetchone()[0]
    cur.execute("SELECT user_id FROM participants WHERE giveaway_id = ?", (gid,))
    users = [r[0] for r in cur.fetchall()]
    if len(users) <= n:
        return users
    return random.sample(users, n)  # без повторов
random.sample даёт уникальных победителей — если нужно 3 разных человека, это правильный выбор.

# scheduler.py
import threading, time
from datetime import datetime

def scheduler_loop():
    while True:
        now = datetime.now().isoformat()
        cur.execute(
            "SELECT id, channel_id, message_id, winners_count FROM giveaways WHERE status='active' AND end_at <= ?",
            (now,)
        )
        for gid, ch, mid, _ in cur.fetchall():
            winners = pick_winners(gid)
            if winners:
                mentions = "\n".join(f"• <a href='tg://user?id={w}'>участник {w}</a>" for w in winners)
                text = f"🎉 <b>Итоги розыгрыша #{gid}</b>\n\nПобедители:\n{mentions}"

Что даёт Telegram для персональных ссылок
Метод bot.create_chat_invite_link() создаёт дополнительную ссылку на чат/канал. Ключевые параметры :

chat_id — канал, куда приглашаем

name — имя ссылки (до 32 символов), можно использовать для метки «кто пригласил»

member_limit — максимум участников, которые могут войти по этой ссылке (1–99999)

expire_date — когда ссылка протухнет (Unix timestamp)

Важно: бот должен быть администратором канала с правом приглашать пользователей .

Схема с персональными ссылками
Как засчитывать приглашения
Проблема: Telegram не присылает боту событие «пользователь X вошёл в канал по ссылке Y». Бот видит только new_chat_members в группах, но для каналов это событие не приходит в том же виде.

Поэтому есть два рабочих подхода:

Вариант А. member_limit = 1 + проверка через get_chat_member
Участник жмёт «Пригласить друзей» в боте.

Бот создаёт ссылку: create_chat_invite_link(channel_id, member_limit=1, name=f"ref_{user_id}_{giveaway_id}").

Участник кидает ссылку другу.

Друг переходит по ссылке и вступает в канал — ссылка «сгорает» (лимит 1).

Бот не получает уведомление о вступлении. Как узнать, что друг зашёл?

Решение: периодически проверять, «сгорела» ли ссылка. Но Telegram не даёт API «проверить, использована ли ссылка». Есть обходной путь: после создания ссылки с member_limit=1, бот может попытаться отозвать её (revoke_chat_invite_link). Если ссылка уже использована — отзыв вернёт ошибку или ссылка уже неактивна. Это ненадёжно.

Более надёжно: при создании ссылки сохранять invite_link в БД. Когда друг заходит в бота (а он должен зайти, чтобы проверить условия или забрать приз), бот просит его нажать /start. В этот момент бот может проверить через get_chat_member(channel_id, friend_id), подписан ли друг на канал. Но как связать «друг пришёл по ссылке участника X»? Никак — Telegram не передаёт эту информацию.

Вариант Б. Ссылка на бота + проверка подписки (рекомендую)
Раз Telegram не даёт отследить «кто кого пригласил в канал», используем ссылку на бота с параметром, а подписку на канал проверяем отдельно.

Участник жмёт «Пригласить друзей» → бот генерирует ссылку:
https://t.me/<bot_username>?start=ref_<user_id>_<giveaway_id>

Друг переходит по ссылке → бот получает /start ref_<user_id>_<giveaway_id> → записывает в БД: invited_id = друг, inviter_id = user_id.

Бот говорит другу: «Чтобы засчитать приглашение, подпишись на канал @xxx».

Друг подписывается → бот проверяет get_chat_member(channel_id, friend_id) → если подписан, засчитывает приглашение участнику.

Плюс: ты точно знаешь, кто кого привёл, и можешь требовать подписку.
Минус: друг должен зайти в бота и нажать /start, а не просто вступить в канал по ссылке. Но для розыгрышей это нормально — участники и так проходят через бота.

Что я рекомендую для твоего бота
Использовать Вариант Б, потому что:

Telegram не даёт надёжного способа отследить вступление в канал по конкретной ссылке.

createChatInviteLink с member_limit=1 создаёт ссылку, но бот не узнает, что она использована .

Ссылка на бота с ?start=ref_... — стандартный и надёжный паттерн для реферальных механик в Telegram.

Но если ты хочешь, чтобы друг вступал именно в канал по ссылке, а не в бота, можно сделать комбинированно:

Бот создаёт персональную ссылку в канал: create_chat_invite_link(channel_id, member_limit=1, name=f"ref_{user_id}").

Бот отправляет участнику эту ссылку + инструкцию: «Перешли другу. Он должен зайти в канал и написать боту /start ref_<user_id>».

Друг заходит в канал по ссылке (лимит 1 сгорает) и обязательно пишет боту /start ref_<user_id>.

Бот проверяет get_chat_member(channel_id, friend_id) → если подписан, засчитывает.

Так ты и ссылку в канал даёшь (красиво), и отслеживаешь приглашение (через бота).

Проверка подписки на каналы делается через метод Bot API getChatMember. Это единственный надёжный способ, и он требует одного обязательного условия: бот должен быть администратором в каждом проверяемом канале .

🔍 Как это работает
Бот отправляет getChatMember(chat_id=@channel, user_id=123). В ответ приходит объект со статусом пользователя. Подписанным считаем, если статус один из:

creator (владелец)

administrator (админ)

member (обычный подписчик)

Всё остальное (left, kicked) — не подписан .

🛠 Как встроить в твой бот (на примере pyTelegramBotAPI / telebot)
Так как мы обсуждали telebot, вот функция проверки, которую можно вызывать в обработчике кнопки «Участвовать» или «Проверить подписку»:

python
def check_subscriptions(bot, user_id, channels):
    """Возвращает список каналов, на которые юзер НЕ подписан."""
    not_subbed = []
    for ch in channels:
        try:
            member = bot.get_chat_member(ch, user_id)
            if member.status not in ('creator', 'administrator', 'member'):
                not_subbed.append(ch)
        except Exception:
            # Если бот не админ или канал недоступен — считаем, что не подписан
            not_subbed.append(ch)
    return not_subbed

Шаг «Подписка на каналы» при создании розыгрыша
Что нужно от админа
Для каждого канала нужны две вещи:

chat_id — для get_chat_member (проверка подписки).

Ссылка — для кнопки-ссылки (https://t.me/... или invite-ссылка для приватного).

Для публичных каналов ссылка строится автоматически из username: @channel → https://t.me/channel. Для приватных — нужна invite-ссылка, и её надо сохранить.

Формат ввода (чтобы не запутаться)
Предлагаю такой сценарий:

Бот: «Пришли каналы, на которые нужно подписаться. Формат:

text
@channel1
@channel2
Для приватного канала: -1001234567890 | https://t.me/+AbCdEf
Каждый канал — с новой строки. Когда закончишь — напиши /done»

Почему так:

Публичный канал → одна строка @channel, ссылка строится сама.

Приватный → chat_id | invite_link, обе части сохраняются.

/done — явный конец ввода, чтобы не гадать, закончил ли админ.

def parse_channel_line(line):
    line = line.strip()
    if '|' in line:
        chat_id, invite_link = [x.strip() for x in line.split('|', 1)]
        if not invite_link.startswith('https://t.me/'):
            return None, "Ссылка должна начинаться с https://t.me/"
        return {'chat_id': chat_id, 'invite_link': invite_link}, None
    else:
        if not line.startswith('@'):
            return None, "Публичный канал должен начинаться с @"
        return {'chat_id': line, 'invite_link': f"https://t.me/{line[1:]}"}, None

Собрал полный проект. Бот на pyTelegramBotAPI, SQLite, два админа через ADMINS = [id1, id2]. Всё в одном файле — проще запустить, потом можно разнести по модулям.

Перед запуском:

pip install pytelegrambotapi

Вписать BOT_TOKEN, ADMINS, CHANNEL_ID (канал, куда публикуются розыгрыши).

Бот должен быть админом в канале публикации и во всех каналах, на которые проверяется подписка (иначе get_chat_member вернёт ошибку).

python
import sqlite3
import random
import threading
import time
import json
from datetime import datetime

import telebot
from telebot import types

# ============ КОНФИГ ============
BOT_TOKEN = "ВСТАВЬ_ТОКЕН"
ADMINS = [111111111, 222222222]   # два админа
CHANNEL_ID = "@your_channel"      # канал для публикации розыгрышей

bot = telebot.TeleBot(BOT_TOKEN)

# ============ БД ============
conn = sqlite3.connect("bot.db", check_same_thread=False)
cur = conn.cursor()

cur.executescript("""
CREATE TABLE IF NOT EXISTS users (
    user_id INTEGER PRIMARY KEY,
    username TEXT,
    verified INTEGER DEFAULT 0,
    first_seen_at TEXT DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS giveaways (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    status TEXT DEFAULT 'draft',          -- draft / active / finished / deleted
    prize_text TEXT,
    prize_photo TEXT,
    winners_count INTEGER,
    required_channels TEXT,               -- JSON list [{"chat_id":..., "invite_link":...}]
    invite_count INTEGER DEFAULT 0,
    no_conditions INTEGER DEFAULT 0,
    end_at TEXT,
    channel_id TEXT,
    message_id INTEGER
);

CREATE TABLE IF NOT EXISTS participants (
    user_id INTEGER NOT NULL,
    giveaway_id INTEGER NOT NULL,
    joined_at TEXT DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, giveaway_id)
);

CREATE TABLE IF NOT EXISTS referrals (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    giveaway_id INTEGER NOT NULL,
    inviter_id INTEGER NOT NULL,
    invited_id INTEGER NOT NULL,
    is_new_user INTEGER DEFAULT 1,
    status TEXT DEFAULT 'pending',        -- pending / confirmed
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    confirmed_at TEXT,
    UNIQUE(giveaway_id, invited_id)
);
""")
conn.commit()

# ============ УТИЛИТЫ ============
def is_admin(uid):
    return uid in ADMINS

def is_verified(uid):
    cur.execute("SELECT verified FROM users WHERE user_id=?", (uid,))
    r = cur.fetchone()
    return bool(r and r[0])

def ensure_user(uid, username=""):
    cur.execute("INSERT OR IGNORE INTO users (user_id, username) VALUES (?,?)", (uid, username))
    conn.commit()

def user_exists(uid):
    cur.execute("SELECT 1 FROM users WHERE user_id=?", (uid,))
    return cur.fetchone() is not None

def mark_verified(uid):
    cur.execute("UPDATE users SET verified=1 WHERE user_id=?", (uid,))
    conn.commit()

def get_giveaway(gid):
    cur.execute("SELECT * FROM giveaways WHERE id=?", (gid,))
    row = cur.fetchone()
    if not row:
        return None
    cols = [d[0] for d in cur.description]
    return dict(zip(cols, row))

def get_active_giveaways():
    cur.execute("SELECT id, prize_text FROM giveaways WHERE status='active'")
    return cur.fetchall()

def is_participant(uid, gid):
    cur.execute("SELECT 1 FROM participants WHERE user_id=? AND giveaway_id=?", (uid, gid))
    return cur.fetchone() is not None

def add_participant(uid, gid):
    cur.execute("INSERT OR IGNORE INTO participants (user_id, giveaway_id) VALUES (?,?)", (uid, gid))
    conn.commit()

def check_subscriptions(uid, channels):
    """channels = [{'chat_id':..., 'invite_link':...}]"""
    not_subbed = []
    for ch in channels:
        try:
            m = bot.get_chat_member(ch["chat_id"], uid)
            if m.status not in ("creator", "administrator", "member"):
                not_subbed.append(ch)
        except Exception:
            not_subbed.append(ch)
    return not_subbed

def confirmed_invites(uid, gid):
    cur.execute("""
        SELECT COUNT(*) FROM referrals
        WHERE inviter_id=? AND giveaway_id=? AND status='confirmed' AND is_new_user=1
    """, (uid, gid))
    return cur.fetchone()[0]

    except Exception:
        return iso

def send_captcha(chat_id):
    a, b = random.randint(1, 10), random.randint(1, 10)
    correct = a + b
    wrong = set()
    while len(wrong) < 3:
        w = random.randint(2, 20)
        if w != correct:
            wrong.add(w)
    options = list(wrong) + [correct]
    random.shuffle(options)

    markup = types.InlineKeyboardMarkup(row_width=2)
    markup.add(*[
        types.InlineKeyboardButton(str(o), callback_data=f"cap:{correct}:{o}")
        for o in options
    ])
    bot.send_message(
        chat_id,
        f"🧮 <b>Реши пример, чтобы продолжить:</b>\n\n<b>{a} + {b} = ?</b>",
        parse_mode="HTML",
        reply_markup=markup
    )

@bot.message_handler(commands=["start"])
def start_cmd(message):
    uid = message.from_user.id
    args = message.text.split(maxsplit=1)
    payload = args[1] if len(args) > 1 else None

    is_new = not user_exists(uid)
    ensure_user(uid, message.from_user.username or "")

    if payload and payload.startswith("ref_"):
        try:
            _, inviter_id, giveaway_id = payload.split("_")
            inviter_id, giveaway_id = int(inviter_id), int(giveaway_id)
            if inviter_id == uid:
                bot.send_message(message.chat.id, "Нельзя пригласить самого себя.")
            elif not is_new:
                bot.send_message(
                    message.chat.id,
                    "👋 Ты уже зарегистрирован в боте раньше.\n"
                    "Приглашение не будет засчитано как «новый друг»."
                )
            else:
                cur.execute(
                    "INSERT OR IGNORE INTO referrals (giveaway_id, inviter_id, invited_id, is_new_user) VALUES (?,?,?,1)",
                    (giveaway_id, inviter_id, uid)
                )
                conn.commit()
        except ValueError:
            pass

    if not is_verified(uid):
        send_captcha(message.chat.id)
        return

    # если есть pending-реферал — сначала подписка
    if try_confirm_pending_referral(message.chat.id, uid):
        return

    show_main_menu(message.chat.id)

def try_confirm_pending_referral(chat_id, uid):
    cur.execute("""
        SELECT id, giveaway_id FROM referrals
        WHERE invited_id=? AND status='pending'
        ORDER BY id DESC LIMIT 1
    """, (uid,))
    row = cur.fetchone()
    if not row:
        return False
    ref_id, gid = row
    gw = get_giveaway(gid)
    if not gw or gw["status"] != "active":
        cur.execute("UPDATE referrals SET status='pending' WHERE id=?", (ref_id,))
        conn.commit()
        return False

    channels = json.loads(gw["required_channels"] or "[]")
    if not channels:
        confirm_referral(ref_id)
        bot.send_message(chat_id, "✅ Приглашение засчитано! Добро пожаловать!")
        return True

    not_subbed = check_subscriptions(uid, channels)
    if not_subbed:
        send_sub_prompt(chat_id, gid, not_subbed, ref_id=ref_id)
        return True

    confirm_referral(ref_id)
    bot.send_message(chat_id, "✅ Приглашение засчитано! Добро пожаловать!")
    return True

def confirm_referral(ref_id):
    cur.execute(
        "UPDATE referrals SET status='confirmed', confirmed_at=? WHERE id=?",
        (datetime.now().isoformat(), ref_id)
    )
    conn.commit()

def init_db():
    cur.executescript("""
    CREATE TABLE IF NOT EXISTS users (
        user_id INTEGER PRIMARY KEY,
        username TEXT,
        verified INTEGER DEFAULT 0,
        first_seen_at TEXT DEFAULT CURRENT_TIMESTAMP
    );
    CREATE TABLE IF NOT EXISTS giveaways (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        status TEXT DEFAULT 'draft',
        prize_text TEXT,
        prize_photo TEXT,
        winners_count INTEGER,
        required_channels TEXT,
        invite_count INTEGER DEFAULT 0,
        no_conditions INTEGER DEFAULT 0,
        end_at TEXT,
        channel_id TEXT,
        message_id INTEGER
    );
    CREATE TABLE IF NOT EXISTS participants (
        user_id INTEGER NOT NULL,
        giveaway_id INTEGER NOT NULL,
        joined_at TEXT DEFAULT CURRENT_TIMESTAMP,
        PRIMARY KEY (user_id, giveaway_id)
    );
    CREATE TABLE IF NOT EXISTS referrals (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        giveaway_id INTEGER NOT NULL,
        inviter_id INTEGER NOT NULL,
        invited_id INTEGER NOT NULL,
        is_new_user INTEGER DEFAULT 1,
        status TEXT DEFAULT 'pending',
        created_at TEXT DEFAULT CURRENT_TIMESTAMP,
        confirmed_at TEXT,
        UNIQUE(giveaway_id, invited_id)
    );
    """)
    conn.commit()

def is_verified(uid):
    cur.execute("SELECT verified FROM users WHERE user_id=?", (uid,))
    r = cur.fetchone()
    return bool(r and r[0])

def ensure_user(uid, username=""):
    cur.execute("INSERT OR IGNORE INTO users (user_id, username) VALUES (?,?)", (uid, username))
    conn.commit()

def user_exists(uid):
    cur.execute("SELECT 1 FROM users WHERE user_id=?", (uid,))
    return cur.fetchone() is not None

def mark_verified(uid):
    cur.execute("UPDATE users SET verified=1 WHERE user_id=?", (uid,))
    conn.commit()

def get_giveaway(gid):
    cur.execute("SELECT * FROM giveaways WHERE id=?", (gid,))
    row = cur.fetchone()
    if not row:
        return None
    cols = [d[0] for d in cur.description]
    return dict(zip(cols, row))

def get_active_giveaways():
    cur.execute("SELECT id, prize_text FROM giveaways WHERE status='active'")
    return cur.fetchall()

def is_participant(uid, gid):
    cur.execute("SELECT 1 FROM participants WHERE user_id=? AND giveaway_id=?", (uid, gid))
    return cur.fetchone() is not None

def add_participant(uid, gid):
    cur.execute("INSERT OR IGNORE INTO participants (user_id, giveaway_id) VALUES (?,?)", (uid, gid))
    conn.commit()

def confirmed_invites(uid, gid):
    cur.execute("""
        SELECT COUNT(*) FROM referrals
        WHERE inviter_id=? AND giveaway_id=? AND status='confirmed' AND is_new_user=1
    """, (uid, gid))
    return cur.fetchone()[0]

def confirm_referral(ref_id):
    from datetime import datetime
    cur.execute(
        "UPDATE referrals SET status='confirmed', confirmed_at=? WHERE id=?",
        (datetime.now().isoformat(), ref_id)
    )
    conn.commit()

def build_captcha_markup(correct, options):
    markup = types.InlineKeyboardMarkup(row_width=2)
    markup.add(*[
        types.InlineKeyboardButton(str(o), callback_data=f"cap:{correct}:{o}")
        for o in options

def check_subscriptions(bot, uid, channels):
    not_subbed = []
    for ch in channels:
        try:
            m = bot.get_chat_member(ch["chat_id"], uid)
            if m.status not in ("creator", "administrator", "member"):
                not_subbed.append(ch)
        except Exception:
            not_subbed.append(ch)
    return not_subbed

def channel_label(ch):
    return ch["chat_id"] if ch["chat_id"].startswith("@") else "Приватный канал"

    channels = json.loads(gw["required_channels"] or "[]")
    lines = []
    if channels:

    if gw["invite_count"]:
        lines.append(f"• Пригласить друзей: {gw['invite_count']}")
    if not lines:
        lines.append("• Нет условий")
    return lines

import threading
import time
import random
from datetime import datetime
from database import cur, conn, get_giveaway
import texts

def finish_giveaway(bot, gid):
    gw = get_giveaway(gid)
    if not gw or gw["status"] != "active":
        return

    cur.execute("SELECT user_id FROM participants WHERE giveaway_id=?", (gid,))
    users = [r[0] for r in cur.fetchall()]
    need = gw["winners_count"]

    if not users:
        bot.send_message(
            gw["channel_id"],
            f"😔 Розыгрыш #{gid} — {gw['prize_text']} завершён.\n\n"
            f"К сожалению, участников не было. Приз остаётся у организатора.",
            reply_to_message_id=gw["message_id"]
        )
    else:
        winners = users if len(users) <= need else random.sample(users, need)
        mentions = []
        for uid in winners:
            cur.execute("SELECT username FROM users WHERE user_id=?", (uid,))
            row = cur.fetchone()
            uname = row[0] if row else None
            mentions.append(texts.mention(uid, uname))

        header = f"🎉 <b>Итоги розыгрыша #{gid} — {gw['prize_text']}</b>\n\n"
        if len(users) < need:
            header += f"🏆 Победители (участников было меньше, чем призов):\n"
        else:
            header += f"🏆 Победители:\n"
        text = header + "\n".join(f"• {m}" for m in mentions) + "\n\nПоздравляем! 🎊"
        bot.send_message(
            gw["channel_id"], text, parse_mode="HTML",
            reply_to_message_id=gw["message_id"]
        )

    cur.execute("UPDATE giveaways SET status='finished' WHERE id=?", (gid,))
    conn.commit()

def scheduler_loop(bot):
    while True:
        try:
            now = datetime.now().isoformat()
            cur.execute(
                "SELECT id FROM giveaways WHERE status='active' AND end_at <= ?", (now,)
            )
            for (gid,) in cur.fetchall():
                finish_giveaway(bot, gid)
        except Exception as e:
            print("scheduler error:", e)
        time.sleep(60)

def start_scheduler(bot):
    t = threading.Thread(target=scheduler_loop, args=(bot,), daemon=True)
    t.start()
